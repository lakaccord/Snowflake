## Snowflake Scripts to create Raw Layer

---

### Switch to SysAdmin Role

```sql
USE ROLE SYSADMIN;
```

### Create Database & Schema

```sql
CREATE OR REPLACE TRANSIENT DATABASE DBT_RAW_DB
COMMENT = 'Database to store Raw data for Demo Project';

CREATE OR REPLACE TRANSIENT SCHEMA JAFFLE_SHOP
COMMENT = 'Schema to store Raw data for jaffle shop';
```

### Create File Format to load CSV files from Stage

```sql
CREATE OR REPLACE FILE FORMAT DBT_RAW_DB.JAFFLE_SHOP.FF_CSV
    TYPE='CSV'
    COMPRESSION = AUTO
    FIELD_DELIMITER = ','
    SKIP_HEADER = 1
    SKIP_BLANK_LINES = TRUE
    DATE_FORMAT = AUTO
    TIME_FORMAT = AUTO
    TIMESTAMP_FORMAT = AUTO
    TRIM_SPACE = TRUE
    FIELD_OPTIONALLY_ENCLOSED_BY = '"'
    ERROR_ON_COLUMN_COUNT_MISMATCH = TRUE
    EMPTY_FIELD_AS_NULL = TRUE
    ENCODING = UTF8;
```

### Create the Stage referencing the S3 Stage location and CSV File Format

```sql
CREATE OR REPLACE STAGE DBT_RAW_DB.JAFFLE_SHOP.RAW_DATA_STAGE
URL = 's3://dbt-tutorial-public/long_term_dataset/'
file_format = DBT_RAW_DB.JAFFLE_SHOP.FF_CSV;

LIST @DBT_RAW_DB.JAFFLE_SHOP.RAW_DATA_STAGE;
```

### Load Raw data into Jaffle Shop schema

#### 1) Create the Raw Customers Table

```sql
CREATE OR REPLACE TRANSIENT TABLE DBT_RAW_DB.JAFFLE_SHOP.RAW_CUSTOMERS
(
    ID VARCHAR(100) PRIMARY KEY,
    NAME VARCHAR(50)
);

-- Load Data into Customers
COPY INTO DBT_RAW_DB.JAFFLE_SHOP.RAW_CUSTOMERS
FROM @DBT_RAW_DB.JAFFLE_SHOP.RAW_DATA_STAGE/raw_customers.csv;

SELECT * FROM DBT_RAW_DB.JAFFLE_SHOP.RAW_CUSTOMERS;
```

#### 2) Create the Raw Stores Table

```sql
CREATE OR REPLACE TRANSIENT TABLE DBT_RAW_DB.JAFFLE_SHOP.RAW_STORES
(
    ID VARCHAR(100) PRIMARY KEY,
    NAME VARCHAR(50),
    OPENED_AT TIMESTAMP_NTZ(9),
    TAX_RATE FLOAT
);

-- Load Data into Stores
COPY INTO DBT_RAW_DB.JAFFLE_SHOP.RAW_STORES
FROM @DBT_RAW_DB.JAFFLE_SHOP.RAW_DATA_STAGE/raw_stores.csv;

SELECT * FROM DBT_RAW_DB.JAFFLE_SHOP.RAW_STORES;
```

#### 3) Create the Raw Orders Table

```sql
CREATE OR REPLACE TRANSIENT TABLE DBT_RAW_DB.JAFFLE_SHOP.RAW_ORDERS
(
    ID VARCHAR(100) PRIMARY KEY,
    CUSTOMER VARCHAR(100) FOREIGN KEY REFERENCES DBT_RAW_DB.JAFFLE_SHOP.RAW_CUSTOMERS(ID),
    ORDERED_AT TIMESTAMP_NTZ(9),
    STORE_ID VARCHAR(100) FOREIGN KEY REFERENCES DBT_RAW_DB.JAFFLE_SHOP.RAW_STORES(ID),
    SUBTOTAL NUMBER(38,0),
    TAX_PAID NUMBER(38,0),
    ORDER_TOTAL NUMBER(38,0)
);

-- Load Data into Orders
COPY INTO DBT_RAW_DB.JAFFLE_SHOP.RAW_ORDERS
FROM @DBT_RAW_DB.JAFFLE_SHOP.RAW_DATA_STAGE/raw_orders.csv;

SELECT * FROM DBT_RAW_DB.JAFFLE_SHOP.RAW_ORDERS;
```

#### 4) Create the Raw Products Table

```sql
CREATE OR REPLACE TRANSIENT TABLE DBT_RAW_DB.JAFFLE_SHOP.RAW_PRODUCTS
(
    SKU VARCHAR(20) PRIMARY KEY,
    NAME VARCHAR(50),
    TYPE VARCHAR(10),
    PRICE NUMBER(38,0),
    DESCRIPTION VARCHAR(500)
);

-- Load Data into Products
COPY INTO DBT_RAW_DB.JAFFLE_SHOP.RAW_PRODUCTS
FROM @DBT_RAW_DB.JAFFLE_SHOP.RAW_DATA_STAGE/raw_products.csv;

SELECT * FROM DBT_RAW_DB.JAFFLE_SHOP.RAW_PRODUCTS;
```

#### 5) Create the Raw Order Items Table

```sql
CREATE OR REPLACE TRANSIENT TABLE DBT_RAW_DB.JAFFLE_SHOP.RAW_ORDER_ITEMS
(
    ID VARCHAR(100) PRIMARY KEY,
    ORDER_ID VARCHAR(100) FOREIGN KEY REFERENCES DBT_RAW_DB.JAFFLE_SHOP.RAW_ORDERS(ID),
    SKU VARCHAR(20) FOREIGN KEY REFERENCES DBT_RAW_DB.JAFFLE_SHOP.RAW_PRODUCTS(SKU)
);

-- Load Data into Order Item
COPY INTO DBT_RAW_DB.JAFFLE_SHOP.RAW_ORDER_ITEMS
FROM @DBT_RAW_DB.JAFFLE_SHOP.RAW_DATA_STAGE/raw_order_items.csv;

SELECT * FROM DBT_RAW_DB.JAFFLE_SHOP.RAW_ORDER_ITEMS;
```

#### 6) Create the Raw Supplies Table

```sql
CREATE OR REPLACE TRANSIENT TABLE DBT_RAW_DB.JAFFLE_SHOP.RAW_SUPPLIES
(
    ID VARCHAR(50) PRIMARY KEY,
    NAME VARCHAR(100),
    COST NUMBER(38,0),
    PERISHABLE BOOLEAN,
    SKU VARCHAR(20) FOREIGN KEY REFERENCES DBT_RAW_DB.JAFFLE_SHOP.RAW_PRODUCTS(SKU)
);

-- Load Data into Supplies
COPY INTO DBT_RAW_DB.JAFFLE_SHOP.RAW_SUPPLIES
FROM @DBT_RAW_DB.JAFFLE_SHOP.RAW_DATA_STAGE/raw_supplies.csv;

SELECT * FROM DBT_RAW_DB.JAFFLE_SHOP.RAW_SUPPLIES;
```
